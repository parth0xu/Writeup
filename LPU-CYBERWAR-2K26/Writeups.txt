def otp_from_state0(s0: int) -> int:
v = (s0 >> 11) & ((1 << 53) - 1)
return (v * M) >> 53

def bounds(n: int):
lo = (n * TWO53 + M - 1) // M
hi = ((n + 1) * TWO53 - 1) // M
return lo, hi

def fwd_int(s0: int, s1: int):
t = s0
s0 = s1
t ^= (t << 23) & MASK
t ^= (t >> 17)
t ^= s0
t ^= (s0 >> 26)
s1 = t & MASK
return s0, s1

t = c0
c0 = c1
t = t ^ ((t << 23) & MASK)
t = t ^ LShR(t, 17)
t = t ^ c0
t = t ^ LShR(c0, 26)
c1 = t & MASK
return c0, c1

x = 0
for i in range(63, -1, -1):
b = (y >> i) & 1
if i + shift <= 63:
b ^= (x >> (i + shift)) & 1
x |= b << i
return x & MASK

def unxorshift_left(y, shift):
x = 0
for i in range(64):
b = (y >> i) & 1
if i - shift >= 0:
b ^= (x >> (i - shift)) & 1
x |= b << i
